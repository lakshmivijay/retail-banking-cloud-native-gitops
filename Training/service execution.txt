docker-compose file
version: "3.8"

services:
  postgres:
    image: postgres:16
    container_name: postgres
    environment:
      POSTGRES_USER: banking
      POSTGRES_PASSWORD: banking
      POSTGRES_DB: customer_db
    volumes:
      - postgres-data:/var/lib/postgresql/data
      - ./init-db.sql:/docker-entrypoint-initdb.d/init-db.sql
    networks:
      - banking-network
    ports:
      - "5432:5432"
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U banking"]
      interval: 5s
      timeout: 5s
      retries: 5

  customer-service:
    build: ./customer-service
    container_name: customer-service
    environment:
      DB_URL: jdbc:postgresql://postgres:5432/customer_db
      DB_USERNAME: banking
      DB_PASSWORD: banking
    depends_on:
      postgres:
        condition: service_healthy
    networks:
      - banking-network
    ports:
      - "8081:8081"

  account-service:
    build: ./account-service
    container_name: account-service
    environment:
      DB_URL: jdbc:postgresql://postgres:5432/account_db
      DB_USERNAME: banking
      DB_PASSWORD: banking
    depends_on:
      postgres:
        condition: service_healthy
    networks:
      - banking-network
    ports:
      - "8082:8082"

  frontend:
    build: ./frontend
    container_name: frontend
    depends_on:
      - customer-service
      - account-service
    networks:
      - banking-network
    ports:
      - "5173:80"

networks:
  banking-network:
    driver: bridge
